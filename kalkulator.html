    <script>

        function hitungHargaMarketing() {

            let pKardus = parseFloat(document.getElementById('p_kardus').value) || 0;

            let lKardus = parseFloat(document.getElementById('l_kardus').value) || 0;

            let tKardus = parseFloat(document.getElementById('t_kardus').value) || 0;

            let pBahan = parseFloat(document.getElementById('p_bahan').value) || 0;

            let lBahan = parseFloat(document.getElementById('l_bahan').value) || 0;

            let pcsPerLbr = parseFloat(document.getElementById('pcs_per_lbr').value) || 0;

            let qtyPesan = parseFloat(document.getElementById('qty_pesan').value) || 0;

            

            let selectFlute = document.getElementById('jenis_flute');

            let faktorPengali = parseFloat(selectFlute.value);

            let namaFlute = selectFlute.options[selectFlute.selectedIndex].text.split(' ')[0]; 


            if(pKardus === 0 || lKardus === 0 || tKardus === 0 || pBahan === 0 || lBahan === 0 || pcsPerLbr === 0 || qtyPesan === 0) {

                alert("Mohon isi semua kolom data termasuk jumlah pesanan (Qty) dengan benar!");

                return;

            }


            // 1. Hitung HPP Bahan Dasar

            let hppBahanDasar = (pBahan * lBahan * faktorPengali) / pcsPerLbr;


            // 2. Ditambah Biaya Tenaga Kerja Rp 120 terlebih dahulu

            let subtotal1 = hppBahanDasar + 120;


            // 3. Tentukan Margin Berjenjang: Jika subtotal1 (HPP + 120) > 1000 pakai 25%, jika <= 1000 pakai 30%

            let persentaseMargin = subtotal1 > 1000 ? 0.25 : 0.30;

            let nilaiMargin = subtotal1 * persentaseMargin;

            let subtotal2 = subtotal1 + nilaiMargin;

            

            // 4. Ditambah Biaya Operasional 15% di tahap akhir

            let biayaOperasional = subtotal2 * 0.15; 

            let hargaFinalPcs = Math.round(subtotal2 + biayaOperasional);


            // Hitung subtotal awal sebelum diskon

            let totalAwal = hargaFinalPcs * qtyPesan;

            let totalAkhir = totalAwal;


            // Logika diskon otomatis jika total di atas Rp 10.000.000

            let infoDiskonEl = document.getElementById('teksDiskon');

            if (totalAwal > 10000000) {

                let nilaiDiskon = totalAwal * 0.05;

                totalAkhir = totalAwal - nilaiDiskon;

                infoDiskonEl.style.display = 'block';

                infoDiskonEl.innerText = `🎉 Diskon 5% diterapkan (-Rp ${Math.round(nilaiDiskon).toLocaleString('id-ID')})`;

            } else {

                infoDiskonEl.style.display = 'none';

            }


            document.getElementById('teksKet').innerText = `${pKardus}x${lKardus}x${tKardus} mm\nBahan ${namaFlute} (K125/M125/K125)`;

            document.getElementById('teksHarga').innerText = "Rp " + hargaFinalPcs.toLocaleString('id-ID');

            

            // Menampilkan rincian perkalian Qty x Harga Satuan serta total akhir setelah diskon

            document.getElementById('teksRincian').innerText = `${qtyPesan.toLocaleString('id-ID')} pcs × Rp ${hargaFinalPcs.toLocaleString('id-ID')}`;

            document.getElementById('teksTotal').innerText = "Rp " + Math.round(totalAkhir).toLocaleString('id-ID');

            

            document.getElementById('hasil').style.display = 'block';

        }

    </script>
